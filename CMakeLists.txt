cmake_minimum_required(VERSION 3.16)
# version.ini deliberately contains only the version number.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/version.ini"
  "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE"
  "${CMAKE_CURRENT_SOURCE_DIR}/packaging/mjd-psu-monitor.svg")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/version.ini" APP_VERSION_FILE)
string(STRIP "${APP_VERSION_FILE}" APP_VERSION_FILE)
if(NOT APP_VERSION_FILE MATCHES "^[0-9]+[.][0-9]+[.][0-9]+$")
  message(FATAL_ERROR "version.ini must contain only a version like 1.2.0")
endif()
project(mjd_psu_monitor VERSION "${APP_VERSION_FILE}" LANGUAGES C)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" APP_LICENSE_C)
string(REPLACE "\\" "\\\\" APP_LICENSE_C "${APP_LICENSE_C}")
string(REPLACE "\"" "\\\"" APP_LICENSE_C "${APP_LICENSE_C}")
string(REPLACE "\r" "" APP_LICENSE_C "${APP_LICENSE_C}")
string(REPLACE "\n" "\\n" APP_LICENSE_C "${APP_LICENSE_C}")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/packaging/mjd-psu-monitor.svg" APP_ICON_SVG_C)
string(REPLACE "\\" "\\\\" APP_ICON_SVG_C "${APP_ICON_SVG_C}")
string(REPLACE "\"" "\\\"" APP_ICON_SVG_C "${APP_ICON_SVG_C}")
string(REPLACE "\r" "" APP_ICON_SVG_C "${APP_ICON_SVG_C}")
string(REPLACE "\n" "\\n" APP_ICON_SVG_C "${APP_ICON_SVG_C}")
configure_file(cmake/app_config.h.in generated/app_config.h @ONLY)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
pkg_check_modules(AYATANA REQUIRED IMPORTED_TARGET ayatana-appindicator3-0.1)
# libsensors does not provide a pkg-config file on every distribution.
find_path(SENSORS_INCLUDE_DIR NAMES sensors/sensors.h)
find_library(SENSORS_LIBRARY NAMES sensors)
if(NOT SENSORS_INCLUDE_DIR OR NOT SENSORS_LIBRARY)
  message(FATAL_ERROR "libsensors development files missing. Run ./prep.sh")
endif()
add_executable(mjd-psu-monitor src/main.c)
target_include_directories(mjd-psu-monitor PRIVATE ${SENSORS_INCLUDE_DIR} "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(mjd-psu-monitor PRIVATE PkgConfig::GTK3 PkgConfig::AYATANA ${SENSORS_LIBRARY})
target_compile_options(mjd-psu-monitor PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS mjd-psu-monitor RUNTIME DESTINATION bin)

install(FILES packaging/mjd-psu-monitor.desktop DESTINATION share/applications)
install(FILES packaging/mjd-psu-monitor.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES LICENSE README.md DESTINATION share/doc/mjd-psu-monitor)
install(FILES LICENSE DESTINATION share/doc/mjd-psu-monitor RENAME copyright)
